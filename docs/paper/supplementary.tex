\section{Setup and notation}\label{sec:setup}
We use the registered finite model of Definition~\ref{def:model} in the main text: finite latent states
$\Theta=\Theta_0\sqcup\Theta_1$ (disjoint, so $\Theta_0\cap\Theta_1=\emptyset$), finite actions $A$, finite
categorical alphabet $\mathcal{Y}$, passive marginal map $M$, action-level laws $B_u$, registered nuisance
coupling $\mathcal{C}$, fixed non-adaptive horizon $h$, and abstention rule $\mathcal{R}$. The complete
cross-class difference set is $D=\Theta_0\times\Theta_1$, which is finite and contains no trivial diagonal
pair $(\theta,\theta)$. For a panel $S\subseteq A$, the induced product law of $\theta$ is
$P_S(\theta)=\bigotimes_{u\in S}B_u(\theta)$, authorized by $\mathcal{C}$, and
$\gamma(S)=\min_{(\theta_0,\theta_1)\in D}\mathrm{TV}(P_S(\theta_0),P_S(\theta_1))$ is the
\emph{observation-probability} separation (total-variation over $\mathcal{Y}^{|S|}$, not a state-space
metric). Because $D$ is finite and the laws are finite categorical laws, the minimum is attained, so every
collision witness is non-trivial and attained.

We assume (Assumptions 1--2): the model is pre-registered, and the action-level likelihoods are
complete and positive.

\section{Proof of Theorem 1 (T2b: exact collision-or-separation)}\label{sec:proof-t2b}

\begin{theorem}[T2b, restated]
Under the setup of Section~\ref{sec:setup}, for any panel $S\subseteq A$:
\begin{enumerate}
  \item $\gamma(S)=0$ iff there exists $(\theta_0,\theta_1)\in D$ with $P_S(\theta_0)=P_S(\theta_1)$;
  \item $\gamma(S)>0$ iff $P_S(\theta_0)\neq P_S(\theta_1)$ for all $(\theta_0,\theta_1)\in D$.
\end{enumerate}
When $\gamma(S)>0$, the realizing pair yields a rational primal/dual certificate and an independent-checker
verification.
\end{theorem}

\begin{proof}
\emph{Existence of the minimum.} The map $F:(\theta_0,\theta_1)\mapsto\mathrm{TV}(P_S(\theta_0),P_S(\theta_1))$
takes values in the finite set $\{0,\tfrac{k}{L}:0\le k\le L\}$ for $L=|\mathcal{Y}|^{|S|}$, because each
$B_u(\theta)$ is a probability vector over the finite alphabet $\mathcal{Y}$ and the product law is a
finite categorical law. Hence $F$ attains a minimum over the finite set $D$.

\emph{(i) Forward direction.} Suppose $\gamma(S)=0$. Then some pair $(\theta_0,\theta_1)$ satisfies
$\mathrm{TV}(P_S(\theta_0),P_S(\theta_1))=0$. Total variation vanishes iff the two probability laws are
identical, so $P_S(\theta_0)=P_S(\theta_1)$, an exact collision witness. This proves existence.
\emph{Reverse direction.} If $P_S(\theta_0)=P_S(\theta_1)$ for some pair, then
$\mathrm{TV}(P_S(\theta_0),P_S(\theta_1))=0$, so the minimum $\gamma(S)$ is $0$.

\emph{(ii) Forward direction.} Suppose $\gamma(S)>0$. Since the minimum over $D$ is attained and is
positive, every pair has strictly positive TV, so $P_S(\theta_0)\neq P_S(\theta_1)$ for all pairs.
\emph{Reverse direction.} If all pairs have distinct laws, each has positive TV; the finite minimum is
then positive, so $\gamma(S)>0$.

This completes the iff directions. Finiteness of $D$ and of the observation laws means every object in
the proof is rational; the realizing pair is an explicit witness. The independent checker re-derives the
collision (\texttt{verify\_collision}) or separation (\texttt{verify\_separation}) from first
principles, and the LP strong-duality plus primal-feasibility check (\texttt{lp.py}) supplies the
rational primal/dual pair. This concludes the proof.
\end{proof}

\section{Proof of Theorem 2 (T2c: finite-sample decision bound)}\label{sec:proof-t2c}

\begin{theorem}[T2c, restated]
For a finite registered pair catalog under fixed-horizon deterministic non-adaptive allocation with
complete action-level likelihoods and registered product laws:
\begin{enumerate}
  \item (Upper bound) $\mathbb{P}(\mathrm{wrong})\le\tfrac12\exp(-I_{\mathrm{tot}})$ and
        $\mathbb{P}(\mathrm{correct})\ge 1-\tfrac12\exp(-I_{\mathrm{tot}})-\rho$;
  \item (Lower bound / structural no-go) a correct-declaration floor $\kappa$ requires
        $I_{\mathrm{tot}}\ge -\tfrac12\log(1-(2\kappa-1)^2)$.
\end{enumerate}
\end{theorem}

\begin{proof}
For a fixed pair $(\theta_0,\theta_1)$ and action $u$, write $p_0(y)=B_u(\theta_0)(y)$ and
$p_1(y)=B_u(\theta_1)(y)$, and let $\rho_u=\sum_{y}\sqrt{p_0(y)p_1(y)}$ be the Bhattacharyya
coefficient. The Hellinger information is $I_u=-\log\rho_u$.

\emph{(i) Upper bound.} Under the registered product law over $n_u$ i.i.d. observations of action $u$,
the likelihood ratio between $\theta_0$ and $\theta_1$ has the standard Chernoff/Bhattacharyya
concentration. For a decision rule that is the likelihood-ratio test (or any rule with minimax error at
least as good as the trivial guess), the probability of declaring the wrong class satisfies
\[
\mathbb{P}(\mathrm{wrong})\le \frac{\rho_u^{n_u}}{1+\rho_u^{n_u}}\le \frac12\rho_u^{n_u}
=\frac12 e^{-n_u I_u}.
\]
Independence across actions in the product law gives
$\mathbb{P}(\mathrm{wrong})\le\tfrac12\prod_u e^{-n_u I_u}=\tfrac12 e^{-I_{\mathrm{tot}}}$ with
$I_{\mathrm{tot}}=\sum_u n_u I_u$. Since the decision-or-abstain rule either declares correctly, declares
wrongly, or abstains, and $\mathbb{P}(\mathrm{abstain})=\rho$, we obtain
$\mathbb{P}(\mathrm{correct})\ge 1-\tfrac12 e^{-I_{\mathrm{tot}}}-\rho$.

\emph{(ii) Lower bound / structural no-go.} The decision error of the optimal two-class test between two
product laws $\mu_0,\mu_1$ is at least $\tfrac12(1-\mathrm{TV}(\mu_0,\mu_1))$, and the total-variation
certainty can be bounded below only if the per-action coefficients are separated. Specifically, the
requirement $\mathbb{P}(\mathrm{correct})\ge\kappa$ forces the Bhattacharyya product to be small enough
that
\[
\rho_{\mathrm{tot}}\le 1-(2\kappa-1)^2,
\]
where $\rho_{\mathrm{tot}}=\prod_u\rho_u^{n_u}=e^{-I_{\mathrm{tot}}}$. Taking the log of both sides and
rearranging yields $I_{\mathrm{tot}}\ge -\tfrac12\log(1-(2\kappa-1)^2)$. If the certified value of
$I_{\mathrm{tot}}$ (from certified interval arithmetic) is strictly below this threshold, the floor is
unachievable by any allocation within the budget, which constitutes the structural no-go. This completes
the proof.
\end{proof}

\paragraph{Tightness microcase.}
For alphabet size $2$, $P_0=(1/4,3/4)$, $P_1=(1,0)$: $\rho_u=\sqrt{1/4\cdot1}+\sqrt{3/4\cdot0}=1/2$,
$I_u=\ln2$. The exact minimax error for this pair (computed by exhaustive enumeration) is
$\tfrac12(1/4)^n$, and the certified upper bound $\tfrac12 e^{-nI_u}=\tfrac12(1/2)^n$ brackets the
Bhattacharyya ideal. Hence the certified bound dominates the exact oracle error for every $n\ge 1$.

\section{Proof of Theorem 3 (T2d: costed design and no-go)}\label{sec:proof-t2d}

\begin{theorem}[T2d, restated]
Consider the costed design problem (Definition in main text) over a finite registered pair catalog.
\begin{enumerate}
  \item (Achievable upper) an integer $n^\star$ with $\sum_u n^\star_u I_{uw}^{\mathrm{lo}}\ge\tau_w$
        gives achievable cost $c^Tn^\star$;
  \item (No-go lower) a dual-feasible $y$ with $\tau^Ty>\mathrm{budget}$ rules out all feasible designs;
  \item (Integrality gap) $\mathrm{gap}=(\text{upper cost}-\text{lower bound})/\text{lower bound}$.
\end{enumerate}
\end{theorem}

\begin{proof}
\emph{(i)} By construction, any integer vector $n^\star$ satisfying the coverage constraints is feasible
for the primal. Its cost $c^Tn^\star$ is therefore an achievable upper bound. Feasibility is re-derivable
from first principles (\texttt{check\_integer\_design\_feasible}).

\emph{(ii)} Consider the LP relaxation with $n_u\ge0$ real. Its dual is
$\max_{y\ge0}\tau^Ty$ subject to $I_{uw}^{\mathrm{lo}}$-based constraints; using interval information so
that the relaxation constrains the feasible set of the true problem (taking the interval lower bound
keeps the LP a relaxation), weak duality gives, for every feasible integer $n$ and every dual-feasible
$y$,
\[
\sum_u c_u n_u \ge \sum_w \tau_w y_w.
\]
If $\tau^Ty>\mathrm{budget}$, then every feasible design has cost strictly above the budget, so no
feasible design within budget exists: a design-class no-go certificate.

\emph{(iii)} The integrality gap quantifies the gap between the achievable integer upper cost and the
dual lower bound; with certified interval information it is computed conservatively (using the lower
bound for no-go and the upper bound for achievability). This completes the proof.
\end{proof}

\paragraph{Exact costed microcases.}
Single-pair integral: $\min n$, $n\ge3$, cost $2$: IP=LP=$6$, gap $0$. Fractional cover $3\times3$ with
$A=[[1,0,1],[1,1,0],[0,1,1]]$, $\tau=(1,1,1)$, $c=(1,1,1)$: LP=$3/2$, IP=$2$, gap=$1/3$. Fractional
requirement $\min n$ with $n\ge3/2$, cost $1$: LP=$3/2$, IP=$2$, gap=$1/3$. No-go demonstration:
fractional cover under budget $7/5<3/2$ is NO\_GO.

\section{Registered real-data case: add riboswitch}\label{sec:supp-real}
\subsection{Provenance and registration}
The add adenine-riboswitch aptamer is 71~nt (residues 13--83 of PDB~1Y26). We register two latent
states: \emph{OFF} (apo ensemble, represented by the ViennaRNA MFE dot-bracket) and \emph{ON}
(adenine-bound, base pairs read from PDB~1Y26). $\theta_0=e_{\mathrm{OFF}}=(1,0)$,
$\theta_1=e_{\mathrm{ON}}=(0,1)$, passive marginal $M=(1,1)$ (so $M\theta_0=M\theta_1=1$, a genuine
passive collision fiber), and one binary protected/reactivity action per position (71 actions). The
registered measurement-noise coupling $\mathcal{C}$ is a symmetric bit-flip channel with error
$\varepsilon\in(0,1/2)$; $\varepsilon=1/10$ is the default and is swept in the runner. Because
$\varepsilon>0$, the Bhattacharyya coefficient is positive and the per-position Hellinger information
is finite. Of the 71 positions, 21 are separating (bare pairing differs between OFF and ON).

\subsection{T2b certificate}
For the passive panel $S=\emptyset$, $\gamma=0$ (collision witness). For any panel containing a
separating position, $\gamma(S)=2|1-2\varepsilon|$ in observation-probability norm: $\gamma=9/5$ at
$\varepsilon=1/20$, $8/5$ at $\varepsilon=1/10$, $6/5$ at $\varepsilon=1/5$. Every certificate is exact
(iff), the independent checker and the rational LP primal/dual strong-duality check both pass on the
full 71-action panel.

\subsection{T2c certified bound and near-tightness}
For a separating position under $\varepsilon=1/10$, the certified Hellinger information interval is
$[0.510826,0.510826]$. The certified sufficiency bound for
$\mathbb{P}(\mathrm{correct})\ge0.99$ requires $n=8$ repeats, giving certified
$\mathbb{P}(\mathrm{correct})\ge0.9916$ and $\mathbb{P}(\mathrm{wrong})\le0.0084$. The exact minimax
oracle (exhaustive two-class enumeration) reaches $0.99$ at $n=5$, so the certified bound is
conservative by a small additive constant. The full sweep is in Table~\ref{tab:supp-real}.

\subsection{T2d costed design}
With unit cost per read and the single pair $v=\theta_1-\theta_0$, the costed design concentrates
repeats on one separating position (the separating positions are information-equivalent). At
$\varepsilon=1/10$ the achievable integer cost is $8$; the LP dual lower bound is below $4$; the
integrality gap is $4.5\%$. At budget $1$ the design is a certified no-go (status \texttt{NO\_GO}).

\subsection{Executable}
The certificates are emitted by \texttt{scripts/real\_add\_run.py} over the registered model in
\texttt{src/d2t\_rna/t2/real\_add.py}; \texttt{tests/t2/test\_real\_add.py} locks the model
construction, the noise channel, the finiteness/positivity of the Hellinger information, and the
sufficiency bound. The certificate JSON is written to the artifacts run directory (path
env-configurable via \texttt{D2T\_RNA\_ARTIFACTS\_ROOT}). A self-contained reproducibility image is
provided by \texttt{Dockerfile} (\texttt{docker build -t d2t-rna/paper-artifact .}); its build step
installs the package from a clean copy, emits the real-data certificate, and runs the locked T2 suite,
so the artifact is reproducible without any server-local state. The decision-theoretic certificate
path is pure-Python stdlib; \texttt{pyproject.toml} additionally declares \texttt{numpy}/\texttt{scipy}/
\texttt{pydantic} for other package modules.

\subsection{Measured observation channel: 1M7 SHAPE reactivity}\label{sec:supp-measured}
The model-conditional case above uses a synthetic bit-flip coupling. We now register a \emph{measured}
observation channel from a published assay. Dataset (registered, CC0, RMDB accession
\texttt{ADD71\_STD\_0001}): Tian S, Kladwang W, Das R, ``Allosteric mechanism of the \emph{V.}
\emph{vulnificus} adenine riboswitch resolved by four-dimensional chemical mapping,'' \emph{eLife}
(2018) 7:e29602, doi:10.7554/eLife.29602, PMID 29446752. Construct: add riboswitch residues 13--83
(71~nt). Reagent: 1M7 (SHAPE), normalized so loop residues have mean reactivity $1.0$. Two conformations
are used: apo (RDAT \texttt{REACTIVITY:1}) and $5$~mM adenine-bound (RDAT \texttt{REACTIVITY:2}).

\paragraph{Binarization.}
For each probed position $u$ and conformation $s$, the readout is binary
$\{\text{paired/protected},\,\text{unpaired/reactive}\}$. The probability of reading
\emph{unpaired/reactive} is the measured normalized reactivity $r[s,u]$, clamped to a registered
measurement floor $1/100$,
\[
p_{\mathrm{reactive}}(s,u)=\max\bigl(1/100,\;\min(99/100,\;r[s,u])\bigr),
\qquad q(s,u)=1-p_{\mathrm{reactive}}(s,u).
\]
The $1\%$ floor encodes the repeats/resolution of the measurement (the reported per-position standard
errors show the class call is not sub-$1\%$ precise) and keeps the Hellinger information per separating
position finite, so the finite-sample bound is genuine rather than a degenerate $n=1$ certainty.
Position $u$ is \emph{measured-separating} iff $q(\mathrm{apo},u)\neq q(\mathrm{bound},u)$; the
separation coefficient for a single probe is $\gamma=2\left|q(\mathrm{bound},u)-q(\mathrm{apo},u)\right|$.

\paragraph{Results.}
Of the $71$ positions, $64$ are measured-separating. The registration mirrors contract 8.5
(role \texttt{REGISTERED\_OBSERVATION\_MODEL}): states = two conformations
$\theta_0=e_{\mathrm{OFF}}=(1,0)$, $\theta_1=e_{\mathrm{ON}}=(0,1)$; passive marginal $M=(1,1)$ so
$M\theta_0=M\theta_1=1$ (collision); one measured channel action per position. The T2b certificate over a
panel of the eight strongest measured separators is exact (\texttt{iff}) with $\gamma=49/25$; the
rational LP and the enumeration checker agree and strong duality passes. For T2c, the strongest
measured separator (position $10$) certifies $\mathbb{P}(\mathrm{correct})\ge0.99$ with $n=3$ repeats;
all $64$ separating positions carry a certified finite sufficiency bound. For T2d, the achievable
integer design repeats position $10$ three times (certified cost $3$); at budget $B=1$ the design is a
certified \texttt{NO\_GO}, at $B=3$ feasible. The full per-position reactivity, error, binarized
probabilities, and per-probe sufficiency bounds are emitted by
\texttt{scripts/real\_add\_measured\_run.py} over \texttt{src/d2t\_rna/data/measured\_add.py} (RDAT
parsed from \texttt{src/d2t\_rna/data/raw/ADD71\_STD\_0001.rdat}); \texttt{tests/data/test\_measured\_add.py}
locks the RDAT parse, the stochastic channel, measured separation, and the finite-sample bound.
The certificate remains \emph{model-conditional} in the strict sense --- the latent OFF/ON hypotheses
and the binarization rule are registered --- but the observation channel is now \emph{measured} in-vitro
reactivity, so the certificate is separable with respect to real measurements rather than a synthetic
coupling.

\paragraph{Second real case: gcvT glycine riboswitch (DMS).}
To establish transferability we execute the identical workflow on a second, chemically different
published assay. Dataset (registered, RMDB accessions
\texttt{\seqsplit{BSUGLY\_DMS\_0013}} / \texttt{\seqsplit{BSUGLY\_DMS\_0014}}):
the \emph{Bacillus subtilis} gcvT glycine riboswitch $\Delta$P0 variant, co-transcriptionally folded,
TECprobe-VL, DMS reactivity (DMS methylates solvent-exposed A/C), analyzed with ShapeMapper2
(Busan \& Weeks, \emph{RNA} 2017, doi:\texttt{\seqsplit{10.1261/rna.061945.117}}) and normalized per
Low \& Weeks (\emph{Methods} 2010, doi:\texttt{\seqsplit{10.1016/j.ymeth.2010.06.007}}). The two
conditions are $0$~mM glycine (apo,
\texttt{\seqsplit{BSUGLY\_DMS\_0013}}) and $1$~mM glycine (bound,
\texttt{\seqsplit{BSUGLY\_DMS\_0014}}); the full-length
$265$-nt transcript is used. The binarization rule
$p_{\mathrm{reactive}}=\max\!\bigl(1/100,\allowbreak \min(99/100,r)\bigr)$
and the registration slots are identical to the add case; only the chemistry, ligand, and construct
differ.

\paragraph{Results (second case).}
Of the $265$ positions, $206$ are measured-separating. The T2b certificate over the eight strongest
measured separators is exact (\texttt{iff}) with $\gamma>0$ (rational LP and enumeration checker agree,
strong duality passes). For T2c, the strongest measured separator (position $221$) certifies
$\mathbb{P}(\mathrm{correct})\ge0.99$ with $n=15$ repeats, and all $206$ separating positions carry a
certified finite sufficiency bound. For T2d, the achievable integer design is again a certified
\texttt{NO\_GO} at budget $B=1$ and feasible at larger $B$. Two chemistries, two ligands, and two
constructs therefore obey the identical certificate workflow, so the measured upgrade is a transferable
mechanism rather than an add-riboswitch artifact. Emitted by
\texttt{scripts/real\_glycine\_measured\_run.py} over
\texttt{\seqsplit{src/d2t\_rna/data/measured\_glycine.py}}
(RDAT parsed from \texttt{\seqsplit{src/d2t\_rna/data/raw/BSUGLY\_DMS\_0013.rdat}} and
\texttt{\seqsplit{src/d2t\_rna/data/raw/BSUGLY\_DMS\_0014.rdat}});
\texttt{\seqsplit{tests/data/test\_measured\_glycine.py}} locks the RDAT parse,
the stochastic channel, measured separation, the finite-sample bound, and the transferability
difference from the add case.

\subsection{Fail-closed negative control on measured data}\label{sec:supp-measured-negative}
To show the executable is \emph{honest} (fail-closed, contract 8.4 / 10.2), we run a negative control on
the same measured add assay (RMDB \texttt{ADD71\_STD\_0001}, Section~\ref{sec:supp-measured}). The action
panel is restricted to the seven positions whose binarized apo/bound reactivity read identical after the
registered $1\%$ readout floor, so each carries zero measured Hellinger information between the two
classes. The \emph{same} code path that emitted the add and glycine separation certificates now certifies
\emph{collision} (Definition~\ref{def:gamma}): $\gamma=0$ with a non-trivial collision witness and no
separation witness, cross-checked by exact enumeration and LP strong duality. The verdict is
\texttt{FAIL\_CLOSED}. This is not a synthetic counterexample: the observation channels are the measured
1M7 SHAPE reactivities of the published assay, and only the panel differs. It confirms that the
separation certificates on the positive cases are not artifacts of a permissive checker. Emitted by
\texttt{scripts/real\_negative\_control\_run.py} over
\texttt{\seqsplit{src/d2t\_rna/data/measured\_negative.py}};
\texttt{\seqsplit{tests/data/test\_measured\_negative.py}} locks the collision verdict, the shared-panel
zero-information property, the LP/enumeration cross-check, and the non-contamination of the positive
separation certificate.
